1. **CDN 加速**：静态资源（图片/CSS/JS）缓存至边缘节点

**冲突检测核心算法**：

```go
// 时间重叠检测（包含缓冲期）
// 新预订：[newStart - buffer, newEnd + buffer]
// 现有预订：[existingBufferStart, existingBufferEnd]

func (s *SlotService) CheckConflict(ctx context.Context, resourceID int64, date time.Time, start, end time.Time, bufferMinutes int) (bool, error) {
    // 计算新预订的缓冲区间
    newBufferStart := start.Add(-time.Duration(bufferMinutes) * time.Minute)
    newBufferEnd := end.Add(time.Duration(bufferMinutes) * time.Minute)

    // 查询所有已预订的槽位（在同一日期、同一资源）
    slots, err := s.repo.GetSlotsByResourceAndDate(ctx, resourceID, date)
    if err != nil {
        return false, err
    }

    for _, slot := range slots {
        if slot.Status != "available" {
            // 重叠检测：现有开始 < 新预订.buffer_end AND 现有结束 > 新预订.buffer_start
            if slot.StartTime.Before(newBufferEnd) && slot.EndTime.After(newBufferStart) {
                return true, nil // 冲突
            }
        }
    }

    return false, nil // 无冲突
}

// 乐观锁更新（防止并发超卖）
func (s *SlotService) LockSlotTx(tx *gorm.DB, slotID int64, bookingID int64, expectedVersion int) (int64, error) {
    result := tx.Model(&TimeSlot{}).
        Where("id = ? AND version = ? AND status = ?", slotID, expectedVersion, "available").
        Updates(map[string]interface{}{
            "status":     "booked",
            "booking_id": bookingID,
            "version":    expectedVersion + 1,
        })

    return result.RowsAffected, result.Error
}
```

-- seckill.lua
local stock_key = KEYS[1]           -- 库存 Key
local user_key = KEYS[2]           -- 用户已购买 Key
local activity_id = ARGV[1]
local user_id = ARGV[2]
local quantity = tonumber(ARGV[3])

-- 1. 检查用户是否已购买（一人一单）
local already_bought = redis.call('SISMEMBER', user_key, user_id)
if already_bought == 1 then
    return -2  -- 错误码：已购买
end

-- 2. 检查库存
local stock = tonumber(redis.call('GET', stock_key))
if stock == nil or stock < quantity then
    return -1  -- 错误码：库存不足
end

-- 3. 原子扣减库存
redis.call('DECRBY', stock_key, quantity)

-- 4. 标记用户已购买
redis.call('SADD', user_key, user_id)

return 1  -- 成功
```

// 秒杀服务核心逻辑
func (s *SeckillService) DoSeckill(ctx context.Context, req *SeckillReq) error {
    // 1. 限流检查
    _, err := sentinel.Entry("seckill")
    if err != nil {
        return errors.New("请求过于频繁")
    }

    // 2. Redis 预扣库存
    result, err := s.redis.Eval(seckillLuaScript, []string{
        fmt.Sprintf("seckill:stock:%d", req.ActivityID),
        fmt.Sprintf("seckill:user:%d", req.ActivityID),
    }, req.ActivityID, req.UserID, 1)

    if result.(int64) != 1 {
        if result.(int64) == -2 {
            return errors.New("您已购买过此活动门票")
        }
        return errors.New("库存不足")
    }

    // 3. 发送事务消息到 MQ
    msg := &SeckillMessage{
        ActivityID: req.ActivityID,
        UserID:     req.UserID,
        TicketCount: 1,
        Timestamp:  time.Now(),
    }

    // 事务消息：本地事务执行成功才发送完整消息
    _, err = s.mq.SendTransactionMessage("seckill_topic", msg,
        func(halfTx func() error) error {
            // 本地事务：创建订单
            return s.createSeckillOrder(msg)
        },
    )

    if err != nil {
        // 本地事务失败，回补 Redis 库存
        s.redis.Incr(fmt.Sprintf("seckill:stock:%d", req.ActivityID))
        return errors.New("订单创建失败")
    }

    return nil
}

// 支付操作（关键：乐观锁 + 延迟 MQ 双重保障）
func (s *OrderService) PayOrder(ctx context.Context, orderID, userID int64) error {
    return s.txManager.WithTransaction(ctx, func(tx *gorm.DB) error {
        // 1. 悲观锁查询（防止并发读取同一订单）
        var order Order
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&order, orderID).Error; err != nil {
            return err
        }

        // 2. 状态校验
        if order.Status != "pending" {
            return errors.New("订单状态不允许支付")
        }

        // 3. 支付截止时间校验
        if time.Now().After(order.PaymentDeadline) {
            return errors.New("订单已超时")
        }

        // 4. 乐观锁更新（版本号检查）
        result := tx.Model(&order).
            Where("version = ?", order.Version).
            Updates(map[string]interface{}{
                "status":    "confirmed",
                "paid_at":   time.Now(),
                "version":   order.Version + 1,
            })

        if result.RowsAffected == 0 {
            return errors.New("并发冲突，请重试")
        }

        // 5. 发送支付成功事件（用于后续流程）
        s.mq.Publish("order_topic", OrderPaidEvent{
            OrderID: order.ID,
            UserID:  order.UserID,
            PaidAt:  time.Now(),
        })

        return nil
    })
}
```

// 延迟队列消费者
type DelayedOrderHandler struct {
    orderRepo *repository.OrderRepository
    mq        *mq.Client
}

func (h *DelayedOrderHandler) HandleMessage(msg *OrderMessage) error {
    order, err := h.orderRepo.GetByID(msg.OrderID)
    if err != nil {
        return err
    }

    // 双重检查：只有在 PENDING 状态才处理
    // 如果已支付（幽灵支付），消息将被忽略
    if order.Status != "pending" {
        return nil
    }

    // 再次检查超时
    if time.Now().After(order.PaymentDeadline) {
        return h.cancelOrder(order, "支付超时自动取消")
    }

    return nil
}
